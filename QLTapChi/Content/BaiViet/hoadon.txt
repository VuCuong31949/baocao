        private void btnin_Click(object sender, EventArgs e)
        {
            // Tạo nội dung hóa đơn từ dữ liệu nhập
            contentToPrint = GenerateInvoiceContent();

        // Tạo nội dung hóa đơn
        private string GenerateInvoiceContent()
        {
            StringBuilder sb = new StringBuilder();
            sb.AppendLine("========== HÓA ĐƠN ==========");
            sb.AppendLine("Khách hàng: " + txtKhachHang.Text);
            sb.AppendLine("Ngày: " + DateTime.Now.ToString("dd/MM/yyyy"));
            sb.AppendLine("-----------------------------------");
            sb.AppendLine("Tên hàng\tSL\tĐơn giá\tThành tiền");

            decimal tongTien = 0;

            foreach (DataGridViewRow row in dgvSanPham.Rows)
            {
                if (row.IsNewRow) continue;

                string ten = row.Cells["TenHang"].Value?.ToString() ?? "";
                string sl = row.Cells["SoLuong"].Value?.ToString() ?? "";
                string gia = row.Cells["DonGia"].Value?.ToString() ?? "";
                string thanhTien = row.Cells["ThanhTien"].Value?.ToString() ?? "";

                sb.AppendLine($"{ten}\t{sl}\t{gia}\t{thanhTien}");

                if (decimal.TryParse(thanhTien, out decimal tien))
                    tongTien += tien;
            }

            sb.AppendLine("-----------------------------------");
            sb.AppendLine("TỔNG CỘNG: " + tongTien.ToString("N0") + " VNĐ");

            return sb.ToString();
        }

        // In nội dung hóa đơn
        private void PrintDocument_PrintPage(object sender, PrintPageEventArgs e)
        {
            Font font = new Font("Arial", 10);
            float y = e.MarginBounds.Top;
            float x = e.MarginBounds.Left;

            e.Graphics.DrawString(contentToPrint, font, Brushes.Black, x, y);
        }
